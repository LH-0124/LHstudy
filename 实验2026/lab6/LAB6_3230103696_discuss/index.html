    1. 核心数学原理 (相似度度量)

    在代码中，我们实现了以下四种相似度指标的手动计算 ：

    • SSD (Sum of Squared Differences, 误差平方和): \(SSD = \sum_{i} (A_i - B_i)^2\)
    • RIU (Ratio Image Uniformity, 比值图像均匀度): \(RIU = \frac{\sigma_R}{\mu_R}\) （其中 \(R_i = \frac{A_i}{B_i}\)，\(\sigma\) 为标准差，\(\mu\) 为均值）
    • CC (Cross-Correlation / Pearson Correlation, 互相关/皮尔逊相关系数): \(CC = \frac{\sum (A_i - \bar{A})(B_i - \bar{B})}{\sqrt{\sum (A_i - \bar{A})^2 \sum (B_i - \bar{B})^2}}\)
    • MI (Mutual Information, 互信息): \(MI = \sum_{a,b} p(a,b) \log\left(\frac{p(a,b)}{p(a)p(b)}\right)\)

    2. 完整 Python 实现代码

    import nibabel as nib
import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import pearsonr
from sklearn.metrics import mutual_info_score

# ==========================================
# 1. 手动实现的相似度计算函数 (基于 Mask)
# ==========================================

def calc_ssd_manual(img1, img2, mask):
    """手动计算 SSD (Sum of Squared Differences)"""
    # 提取 mask 内的体素
    v1 = img1[mask]
    v2 = img2[mask]
    return np.sum((v1 - v2) ** 2)

def calc_riu_manual(img1, img2, mask, epsilon=1e-8):
    """手动计算 RIU (Ratio Image Uniformity)"""
    v1 = img1[mask]
    v2 = img2[mask]
    # 避免除以0
    ratio = v1 / (v2 + epsilon)
    return np.std(ratio) / (np.mean(ratio) + epsilon)

def calc_cc_manual(img1, img2, mask):
    """手动计算 CC (Cross-Correlation)"""
    v1 = img1[mask]
    v2 = img2[mask]

    mean1 = np.mean(v1)
    mean2 = np.mean(v2)

    numerator = np.sum((v1 - mean1) * (v2 - mean2))
    denominator = np.sqrt(np.sum((v1 - mean1)**2) * np.sum((v2 - mean2)**2))

    if denominator == 0:
        return 0
    return numerator / denominator

def calc_mi_manual(img1, img2, mask, bins=256):
    """手动计算 MI (Mutual Information)"""
    v1 = img1[mask]
    v2 = img2[mask]

    # 计算联合直方图
    hist_2d, _, _ = np.histogram2d(v1, v2, bins=bins)

    # 将直方图转换为概率 p(a,b)
    pxy = hist_2d / float(np.sum(hist_2d))

    # 计算边缘概率 p(a) 和 p(b)
    px = np.sum(pxy, axis=1) # 对 y 积分
    py = np.sum(pxy, axis=0) # 对 x 积分

    # 广播计算 p(a)p(b)
    px_py = px[:, None] * py[None, :]

    # 仅计算非零项的 MI 避免 log(0)
    non_zero = pxy > 0
    mi = np.sum(pxy[non_zero] * np.log(pxy[non_zero] / px_py[non_zero]))
    return mi

# ==========================================
# 2. 标准库实现的对比函数
# ==========================================

def calc_metrics_library(img1, img2, mask, bins=256):
    """使用标准库计算 SSD, CC, MI (RIU除外)"""
    v1 = img1[mask]
    v2 = img2[mask]

    # SSD: numpy
    ssd_lib = np.sum((v1 - v2) ** 2)

    # CC: scipy
    cc_lib, _ = pearsonr(v1, v2)

    # MI: scikit-learn
    # sklearn 需要一维直方图的输入，或者自己分箱
    # 离散化数据以匹配 sklearn 的行为
    c1 = np.digitize(v1, np.histogram_bin_edges(v1, bins=bins))
    c2 = np.digitize(v2, np.histogram_bin_edges(v2, bins=bins))
    mi_lib = mutual_info_score(c1, c2)

    return ssd_lib, cc_lib, mi_lib

# ==========================================

# ==========================================

def apply_rigid_transform(img, theta_deg, tx, ty):
    """
    对图像应用刚性变换 (绕图像中心旋转，并平移)
    theta_deg: 旋转角度 (度)
    tx, ty: 像素平移量
    """
    center_y, center_x = np.array(img.shape) / 2.0

    # 构建变换矩阵 (平移到中心 -> 旋转 -> 平移回中心并加上位移)
    t_center = AffineTransform(translation=(-center_x, -center_y))
    rot = AffineTransform(rotation=np.deg2rad(theta_deg))
    t_uncenter_translate = AffineTransform(translation=(center_x + tx, center_y + ty))

    # 组合变换矩阵
    matrix = t_center.params
    matrix = rot.params @ matrix
    matrix = t_uncenter_translate.params @ matrix

    t_final = AffineTransform(matrix=matrix)
    # 使用 inverse 映射，因为 warp 默认使用逆变换采样
    return warp(img, t_final.inverse, preserve_range=True)

def calc_overlap(mask1, mask2):
    """计算图像重叠度 (Dice Coefficient)"""
    intersection = np.logical_and(mask1, mask2)
    return 2. * intersection.sum() / (mask1.sum() + mask2.sum())

def loss_ssd(target, moving, threshold=10):
    mask = (target > threshold) & (moving > threshold)
    if mask.sum() == 0: return float('inf')
    return np.mean((target[mask] - moving[mask]) ** 2)

def loss_riu(target, moving, threshold=10):
    """RIU 用于跨模态 (Inter-modality)，越小越均匀"""
    mask = (target > threshold) & (moving > threshold)
    if mask.sum() == 0: return float('inf')
    ratio = target[mask] / (moving[mask] + 1e-8)
    return np.std(ratio) / (np.mean(ratio) + 1e-8)

def steepest_descent_step(target, moving, current_params, metric_fn, step_size):
    """计算数值梯度并执行一次最速下降"""
    theta, tx, ty = current_params
    delta_theta = 0.5 # 角度微元
    delta_t = 0.5     # 平移微元

    # 计算数值梯度 (中心差分)
    l_theta_plus = metric_fn(target, apply_rigid_transform(moving, theta + delta_theta, tx, ty))
    l_theta_minus = metric_fn(target, apply_rigid_transform(moving, theta - delta_theta, tx, ty))
    grad_theta = (l_theta_plus - l_theta_minus) / (2 * delta_theta)

    l_tx_plus = metric_fn(target, apply_rigid_transform(moving, theta, tx + delta_t, ty))
    l_tx_minus = metric_fn(target, apply_rigid_transform(moving, theta, tx - delta_t, ty))
    grad_tx = (l_tx_plus - l_tx_minus) / (2 * delta_t)

    l_ty_plus = metric_fn(target, apply_rigid_transform(moving, theta, tx, ty + delta_t))
    l_ty_minus = metric_fn(target, apply_rigid_transform(moving, theta, tx, ty - delta_t))
    grad_ty = (l_ty_plus - l_ty_minus) / (2 * delta_t)

    grad = np.array([grad_theta, grad_tx, grad_ty])

    # 科学方法改进：梯度归一化，防止不同 metric 量级导致的步长爆炸
    norm = np.linalg.norm(grad)
    if norm > 1e-8:
        grad = grad / norm

    # 更新参数 (向负梯度方向移动)
    new_params = current_params - step_size * grad
    return new_params

def register_pyramid(target_img, moving_img, metric_fn):
    """使用多尺度金字塔执行配准"""
    # 定义金字塔层级 (缩放因子) 和每层的迭代次数、初始步长
    pyramid_levels = [0.25, 0.5, 1.0]
    iterations_per_level = [30, 20, 20]
    base_step_sizes = np.array([5.0, 2.0, 0.5]) # 步长随分辨率增大而减小

    # 初始猜测: 0 角度, 0 平移
    current_params = np.array([0.0, 0.0, 0.0]) 

    for level_idx, scale in enumerate(pyramid_levels):
        print(f"  -> 金字塔层级: {scale}x")
        # 缩小图像
        t_scaled = rescale(target_img, scale, anti_aliasing=True)
        m_scaled = rescale(moving_img, scale, anti_aliasing=True)

        if level_idx > 0:
            scale_ratio = pyramid_levels[level_idx] / pyramid_levels[level_idx-1]
            current_params[1] *= scale_ratio
            current_params[2] *= scale_ratio

            current_params = steepest_descent_step(
                t_scaled, m_scaled, current_params, metric_fn, current_step

    params = np.array([0.0, 0.0, 0.0]) 

    for level_idx, (scale, num_iters, base_lr) in enumerate(zip(pyramid_levels, iterations_per_level, base_learning_rates)):
        print(f"    -> 正在执行 {scale*100}% 分辨率层级配准...")

        # 下采样图像 (anti_aliasing 避免混叠伪影)
        t_scaled = rescale(target, scale, anti_aliasing=True)
        m_scaled = rescale(moving, scale, anti_aliasing=True)

        # 数值求导的微小偏移量
        d_theta = 0.5
        d_t = 0.5 # 以原始像素尺度衡量的平移微元

        for i in range(num_iters):

            g_theta = (get_loss(theta + d_theta, tx, ty) - get_loss(theta - d_theta, tx, ty)) / (2 * d_theta)

            norm = np.linalg.norm(grad)

def apply_rigid_transform(img, theta_deg, tx, ty):
    """施加刚性变换 (绕图像中心旋转，并平移)"""
    center_y, center_x = np.array(img.shape) / 2.0
    t_center = AffineTransform(translation=(-center_x, -center_y))

    matrix = rot.params @ matrix
    matrix = t_uncenter_translate.params @ matrix

    t_final = AffineTransform(matrix=matrix)

def calc_overlap_dice(mask1, mask2):
    intersection = np.logical_and(mask1, mask2)
    if mask1.sum() + mask2.sum() == 0: return 0
    return 2. * intersection.sum() / (mask1.sum() + mask2.sum())

    h, w = target.shape
    rgb_img = np.zeros((h, w, 3))
    rgb_img[..., 0] = (target > threshold) * 1.0  
    rgb_img[..., 1] = (moving > threshold) * 1.0  
    return rgb_img

# ==========================================

def loss_mi(target, moving, threshold=10, bins=64):
    """
    计算互信息。
    注意：scipy.optimize.minimize 寻找的是最小值，
    互信息越大越好，所以我们返回互信息的负值 (-MI)。
    """
    mask = (target > threshold) & (moving > threshold)
    if mask.sum() == 0: return float('inf')

    v_t = target[mask]
    v_m = moving[mask]

    # 使用直方图计算联合概率分布
    hist_2d, _, _ = np.histogram2d(v_t, v_m, bins=bins)
    pxy = hist_2d / float(np.sum(hist_2d))

    px = np.sum(pxy, axis=1)
    py = np.sum(pxy, axis=0)
    px_py = px[:, None] * py[None, :]

    non_zero = pxy > 0
    mi = np.sum(pxy[non_zero] * np.log(pxy[non_zero] / px_py[non_zero]))

    return -mi  # 负号！

    py = np.sum(pxy, axis=0)

def apply_rigid_transform_skimage(img, theta_deg, tx, ty):
    center_y, center_x = np.array(img.shape) / 2.0
    t_center = AffineTransform(translation=(-center_x, -center_y))
    rot = AffineTransform(rotation=np.deg2rad(theta_deg))
    t_uncenter_translate = AffineTransform(translation=(center_x + tx, center_y + ty))

    matrix = t_center.params
    matrix = rot.params @ matrix
    matrix = t_uncenter_translate.params @ matrix
    t_final = AffineTransform(matrix=matrix)
    return warp(img, t_final.inverse, preserve_range=True, order=1)

def calc_overlap_dice(mask1, mask2):
    intersection = np.logical_and(mask1, mask2)
    if mask1.sum() + mask2.sum() == 0: return 0
    return 2. * intersection.sum() / (mask1.sum() + mask2.sum())

def create_overlap_image(target, moving, threshold=10):
    h, w = target.shape
    rgb_img = np.zeros((h, w, 3))
    rgb_img[..., 0] = (target > threshold) * 1.0  
    rgb_img[..., 1] = (moving > threshold) * 1.0  
    return rgb_img

def loss_ssd(target, moving, threshold=10):
    """【模态内】计算 SSD"""
    mask = target > threshold 
    if mask.sum() == 0: return float('inf')
    return np.mean((target[mask] - moving[mask]) ** 2)

def loss_mi(target, moving, threshold=10, bins=32):
    """【跨模态】计算 MI (互信息，取负值，越小越好)"""
    mask = (target > threshold) & (moving > threshold)
    if mask.sum() < 100: return float('inf')
    v_t = target[mask]
    v_m = moving[mask]
    hist_2d, _, _ = np.histogram2d(v_t, v_m, bins=bins)
    pxy = hist_2d / float(np.sum(hist_2d))
    px = np.sum(pxy, axis=1)
    py = np.sum(pxy, axis=0)
    px_py = px[:, None] * py[None, :]
    non_zero = pxy > 0
    mi = np.sum(pxy[non_zero] * np.log(pxy[non_zero] / (px_py[non_zero] + 1e-8)))
    return -mi

def basic_steepest_descent(target, moving, metric_fn, iterations=120, learning_rate=2.0):

    delta_theta = 0.5 
    delta_t = 0.5

    🔍 为什么使用 RIU 会导致图像“逃跑”？（写进报告绝对加分）

    PPT 明确要求跨模态不能用 MI（互信息），因为 MI 的离散直方图计算方式是阶梯状的，无法求导，SGD 会直接瘫痪。于是我们退而求其次选了 Project 1 中的 RIU（比值图像均匀度）。

    但 RIU 的公式是 \(RIU = \frac{\text{Std}(T1/T2)}{\text{Mean}(T1/T2)}\)。它的逻辑是：两张图越相似，比值越接近常数，方差越小，RIU 越小。我们希望通过最速下降法去寻找 RIU 的最小值。

    灾难发生的数学逻辑： 1. 对齐时的惩罚： T1 和 T2 的灰度是反转的（比如脑脊液在 T1 是黑的，T2 是白的；白质在 T1 是白的，T2 是灰黑的）。当两张脑部图像完美对齐时，\(T1 / T2\) 的比值会剧烈波动（时而极大，时而极小），导致标准差（Std）极大，此时的 RIU 反而非常高。 2. 错位时的奖励： 当 SGD 尝试把 T2 图像往旁边平移时，T1 图像对应的分母变成了黑色的背景（常数，且为了防报错被我们截断为 1）。此时比值 \(T1 / 1\) 就等于 T1 图像本身。T1 图像本身的均匀度（方差），远远小于 T1/T2 错乱的对比度！ 3. 优化器的“偷懒”： 优化器发现：“哇！只要我把 T2 图像推得越远，引入的背景越多，RIU 的值就越小！” 于是它顺着梯度一路狂奔，直接把图像扔出了屏幕。它完美地执行了数学指令，只是这个数学指标在这里失效了。

    💡 真正的救星：平方互相关系数 (\(CC^2\))

    既然 MI 不能用（不可导），RIU 不能用（对比度反转陷阱），Project 1 中唯一剩下的且最适合跨模态的指标就是 CC（互相关/皮尔逊相关系数）。

    虽然 T1 和 T2 灰度反转，导致它们的 CC 值可能是负数（比如 -0.4），或者非完全正相关。但关键在于，当解剖结构对齐时，相关性的绝对值一定是最大的！ 因此，对于跨模态任务，我们明确使用的相似度指标是：负的 CC 平方 (\(-CC^2\))。 求 \(-CC^2\) 的最小值，就等于求相关性强度的最大值。这个曲面极其平滑，SGD 会非常喜欢！

# 1. 核心数学与变换工具

def apply_rigid_transform(img, theta_deg, tx, ty):
    """施加刚性变换 (绕中心旋转，并平移)"""
    center_y, center_x = np.array(img.shape) / 2.0
    t_center = AffineTransform(translation=(-center_x, -center_y))
    rot = AffineTransform(rotation=np.deg2rad(theta_deg))
    t_uncenter_translate = AffineTransform(translation=(center_x + tx, center_y + ty))

    matrix = t_center.params
    matrix = rot.params @ matrix
    matrix = t_uncenter_translate.params @ matrix

    t_final = AffineTransform(matrix=matrix)
    return warp(img, t_final.inverse, preserve_range=True)

def calc_overlap_dice(mask1, mask2):
    """计算图像重叠度 (Dice Coefficient)"""
    intersection = np.logical_and(mask1, mask2)
    if mask1.sum() + mask2.sum() == 0: return 0
    return 2. * intersection.sum() / (mask1.sum() + mask2.sum())

# ==========================================

def calc_ssd(target, moving, threshold=10):
    """【明确指标 1】：模态内使用 SSD (误差平方和)"""
    # 使用静态掩膜，只关注目标区域，防止图像旋转边缘引入的 0 值破坏梯度
    mask = target > threshold
    if mask.sum() == 0: return float('inf')
    return np.mean((target[mask] - moving[mask]) ** 2)

def calc_cc_squared(target, moving, threshold=10):
    """【明确指标 2】：跨模态使用 CC平方的负值 (-CC^2)"""
    # 使用交集掩膜，只计算真正重叠的大脑区域
    mask = (target > threshold) & (moving > threshold)
    if mask.sum() < 500: return 0.0 # 重叠太少直接返回极差值(0)，防止除0错误

    v_t = target[mask]
    v_m = moving[mask]

    m_t = np.mean(v_t)
    m_m = np.mean(v_m)

    num = np.sum((v_t - m_t) * (v_m - m_m))
    den = np.sqrt(np.sum((v_t - m_t)**2) * np.sum((v_m - m_m)**2))

    if den == 0: return 0.0
    cc = num / den

    # 最小化 -CC^2 等同于最大化相关性的绝对强度
    return -(cc ** 2)

def basic_steepest_descent(target, moving, metric_fn, iterations=80, learning_rate=1.0):
    """基础的最速下降法 (不使用金字塔，保持你原有的逻辑结构)"""
    current_params = np.array([0.0, 0.0, 0.0]) # [theta, tx, ty]

    delta_theta = 0.5 
    delta_t = 0.5     

    for i in range(iterations):
        theta, tx, ty = current_params

        # 加入简单的线性衰减，让收敛更平稳
        lr = learning_rate * (1.0 - i / iterations)

        # 计算数值梯度
        l_theta_plus = metric_fn(target, apply_rigid_transform(moving, theta + delta_theta, tx, ty))
        l_theta_minus = metric_fn(target, apply_rigid_transform(moving, theta - delta_theta, tx, ty))
        grad_theta = (l_theta_plus - l_theta_minus) / (2 * delta_theta)

        l_tx_plus = metric_fn(target, apply_rigid_transform(moving, theta, tx + delta_t, ty))
        l_tx_minus = metric_fn(target, apply_rigid_transform(moving, theta, tx - delta_t, ty))
        grad_tx = (l_tx_plus - l_tx_minus) / (2 * delta_t)

        l_ty_plus = metric_fn(target, apply_rigid_transform(moving, theta, tx, ty + delta_t))
        l_ty_minus = metric_fn(target, apply_rigid_transform(moving, theta, tx, ty - delta_t))
        grad_ty = (l_ty_plus - l_ty_minus) / (2 * delta_t)

        grad = np.array([grad_theta, grad_tx, grad_ty])

        # 为了保证基础版的 SGD 能够稳定运行，对梯度方向进行简单的归一化
        norm = np.linalg.norm(grad)
        if norm > 1e-8:
            grad = grad / norm

        # 梯度下降更新
        current_params = current_params - lr * grad

    return current_params
