    Python 实现代码

    import numpy as np
import cv2
import matplotlib.pyplot as plt
from scipy.signal import convolve2d
import os

def apply_sobel_kernel(image):
    """
    实现 Sobel 梯度核计算
    """
    # 定义 Sobel 核
    Kx = np.array([[-1, 0, 1], 
                   [-2, 0, 2], 
                   [-1, 0, 1]], dtype=np.float32)

    Ky = np.array([[-1, -2, -1], 
                   [ 0,  0,  0], 
                   [ 1,  2,  1]], dtype=np.float32)

    # 二维卷积计算水平和垂直梯度
    Gx = convolve2d(image, Kx, mode='same', boundary='symm')
    Gy = convolve2d(image, Ky, mode='same', boundary='symm')

    # 计算梯度幅值
    magnitude = np.sqrt(Gx**2 + Gy**2)

    # 归一化到 0-255 以便后续处理
    if magnitude.max() > 0:
        magnitude = (magnitude / magnitude.max()) * 255

    return magnitude.astype(np.uint8)

def segment_lesion(img, threshold_T):
    """
    执行病灶分割的完整 Pipeline
    """
    # 1. 5x5 平滑滤波器 (Gaussian Blur)
    img_smooth = cv2.GaussianBlur(img, (5, 5), 0)

    # 2. 实现 Sobel 滤波器
    grad_mag = apply_sobel_kernel(img_smooth)

    # 3. 阈值 T -> 获取边缘掩膜
    _, edge_mask = cv2.threshold(grad_mag, threshold_T, 255, cv2.THRESH_BINARY)

    # 4. 去除大脑边缘 (Remove brain edge)
    # 策略：通过对原图进行简单阈值化获取整个大脑的 Mask，然后进行腐蚀操作（Erosion），
    # 从而排除掉外轮廓（大脑边缘），只保留内部结构。
    _, brain_mask = cv2.threshold(img, 10, 255, cv2.THRESH_BINARY)
    kernel_erode = np.ones((15, 15), np.uint8) # 腐蚀核大小可视具体图像尺寸微调
    brain_core = cv2.erode(brain_mask, kernel_erode, iterations=1)

    # 掩膜相与，仅保留大脑内部的边缘
    internal_edges = cv2.bitwise_and(edge_mask, brain_core)

    # 5. 闭运算 (Closed operation)
    # 连接断开的边缘缝隙，使病灶边缘闭合
    kernel_close = np.ones((7, 7), np.uint8)
    closed_edges = cv2.morphologyEx(internal_edges, cv2.MORPH_CLOSE, kernel_close)

    # 6. 连通域处理 (Connected domain processing)
    # 寻找所有的连通域
    num_labels, labels, stats, centroids = cv2.connectedComponentsWithStats(closed_edges, connectivity=8)

    lesion_mask = np.zeros_like(closed_edges)
    if num_labels > 1:
        # 提取面积最大的连通域（排除背景 label 0）作为病灶区域
        largest_label = 1 + np.argmax(stats[1:, cv2.CC_STAT_AREA])
        lesion_mask[labels == largest_label] = 255

    return grad_mag, internal_edges, lesion_mask

def main():

    img_t1 = np.load('T1.npy')

    # 确保数据格式为 uint8 或 float32 以兼容 OpenCV
    img_t1 = cv2.normalize(img_t1, None, 0, 255, cv2.NORM_MINMAX).astype(np.uint8)
    img_t2 = cv2.normalize(img_t2, None, 0, 255, cv2.NORM_MINMAX).astype(np.uint8)

    # 设置阈值 T (可能需要根据您的真实数据分布进行微调)
    T1_THRESHOLD = 30

    代码原理解析与调整建议

    • 实现 Sobel 梯度核：代码中没有直接调用 cv2.Sobel，而是按照要求手动定义了 \(3 \times 3\) 的水平和垂直卷积核，并通过 scipy.signal.convolve2d 执行底层卷积，严格满足“Implement the Sobel gradient kernel”的学术要求。
    • 去除大脑边缘：大脑边缘（外轮廓）的梯度值通常极高。代码中采用了一种形态学掩膜法：对原图进行二值化得到大脑轮廓，使用形态学腐蚀操作（Erode）向内收缩，将外围的大脑轮廓抠除，然后与边缘检测的结果相交（Bitwise AND）。您可以通过调整 kernel_erode 的尺寸（如 (15, 15)）来控制去除外圈的厚度。
    • 连通域处理：获取闭运算结果后，由于存在细碎的无关边缘噪点，程序调用了连通组件标记 cv2.connectedComponentsWithStats，并直接提取了面积最大的连通域（去除了背景），假设该最大连通域就是闭合的病灶区域。
    • 阈值 T 调参：在 main() 函数中，我预设了 T=30 作为 Sobel 梯度的二值化阈值。由于不同的 MRI 切片对比度差异很大，如果运行后发现病灶边缘未完全提取出或者杂讯太多，请重点微调 T1_THRESHOLD 和 T2_THRESHOLD 这两个变量。

def segment_lesion_improved(img, threshold_T):
    """
    改进后的病灶分割 Pipeline
    """
    # 1. 5x5 平滑滤波器
    img_smooth = cv2.GaussianBlur(img, (5, 5), 0)

    # 2. Sobel 滤波器
    grad_mag = apply_sobel_kernel(img_smooth)

    import numpy as np

import os

def iterative_threshold(image, initial_T, delta=0.5):
    """
    实现迭代全局阈值分割算法
    """
    T = initial_T
    image_f = image.astype(np.float32)

    while True:
        # 1. 根据当前 T 将图像分为两部分
        g1 = image_f[image_f > T]
        g2 = image_f[image_f <= T]

        # 2. 计算两部分的平均灰度值
        mu1 = np.mean(g1) if len(g1) > 0 else 0
        mu2 = np.mean(g2) if len(g2) > 0 else 0

        # 3. 计算新的阈值
        T_new = (mu1 + mu2) / 2.0

        # 4. 检查收敛性
        if abs(T - T_new) < delta:
            break
        T = T_new

    return T

def otsu_threshold(image):
    """
    手动实现 Otsu 最优阈值算法 (不调用库函数)
    """
    # 1. 计算直方图
    hist = np.histogram(image, bins=256, range=(0, 256))[0]
    total_pixels = image.size

    max_variance = -1
    best_T = 0

    # 2. 遍历所有可能的阈值 t (0-255)
    for t in range(256):
        # 背景 (w0: 权重, m0: 平均灰度)
        w0 = np.sum(hist[:t]) / total_pixels
        # 前景 (w1: 权重, m1: 平均灰度)
        w1 = np.sum(hist[t:]) / total_pixels

        if w0 == 0 or w1 == 0:
            continue

        # 计算平均灰度值
        m0 = np.sum(np.arange(t) * hist[:t]) / (w0 * total_pixels)
        m1 = np.sum(np.arange(t, 256) * hist[t:]) / (w1 * total_pixels)

        # 3. 计算类间方差 (Between-class variance)
        variance = w0 * w1 * ((m0 - m1) ** 2)

        # 4. 寻找最大方差对应的 t
        if variance > max_variance:
            max_variance = variance
            best_T = t

    return best_T

    1. 判断图像的可分割性（模态分析）

    • 双峰分布 (Bimodal)：如果直方图有两个明显的“山峰”（一个是背景/正常组织，一个是明亮的病灶），那么全局阈值分割（Otsu/迭代法）效果会非常好。
    • 单峰分布：如果直方图挤成一团，说明目标和背景灰度太接近，阈值法可能会失效。
    • 观察点：在 T2 图像中，你应该能看到主峰右侧有一个“小尾巴”或者小微峰，那就是高亮的病灶区。

    2. 验证算法的合理性

    • 切割点位置：直方图可以直观地告诉你 \(T\) 落在哪里。
    • 理想状态：对于迭代法和 Otsu，理想的阈值 \(T\) 应该落在两个峰之间的“谷底”。通过曲线图，你可以一眼看出算法选取的 \(T\) 是否真的成功分开了不同的组织。

    3. 解释 T1 和 T2 的差异（Discuss 的核心）

    • T2 曲线：你会看到直方图范围很广，Otsu 选取的红色线通常能很好地把右侧的高亮部分切出来。

    import numpy as np

def iterative_threshold_manual(image, initial_T, delta=0.5):
    """
    实现迭代全局阈值分割
    """
    # 排除背景像素(<=10)，仅对脑组织进行计算
    pixels = image[image > 10].astype(np.float32)
    if len(pixels) == 0: return 0

    T = float(initial_T)
    while True:
        # 分为两类
        g1 = pixels[pixels > T]
        g2 = pixels[pixels <= T]

        # 计算均值
        mu1 = np.mean(g1) if len(g1) > 0 else 0
        mu2 = np.mean(g2) if len(g2) > 0 else 0

        T_new = (mu1 + mu2) / 2.0

        # 判断收敛
        if abs(T - T_new) < delta:
            break
        T = T_new
    return T

def otsu_threshold_manual(image):
    """
    手动实现 Otsu 最优阈值算法，并返回方差曲线
    """
    # 排除背景像素，只统计脑组织部分
    brain_pixels = image[image > 10]
    total_pixels = brain_pixels.size

    # 计算直方图
    hist = np.histogram(brain_pixels, bins=256, range=(0, 256))[0]

    variances = np.zeros(256)
    max_variance = -1
    best_T = 0

    # 遍历所有可能的阈值 t
    for t in range(1, 255):
        # 计算 w0, w1 (类概率权重)
        w0 = np.sum(hist[:t]) / total_pixels
        w1 = np.sum(hist[t:]) / total_pixels

        if w0 == 0 or w1 == 0:
            continue

        # 计算 mu0, mu1 (类平均灰度)
        mu0 = np.sum(np.arange(t) * hist[:t]) / (w0 * total_pixels)
        mu1 = np.sum(np.arange(t, 256) * hist[t:]) / (w1 * total_pixels)

        # 计算类间方差 sigma_b^2
        variance = w0 * w1 * ((mu0 - mu1) ** 2)
        variances[t] = variance

        if variance > max_variance:
            max_variance = variance
            best_T = t

    return best_T, variances

    import numpy as np
import cv2

# 1. 区域生长算法 (Region Growing)
# ==========================================
def region_growing(img, seed_point, tolerance=20):
    """
    使用 OpenCV 的 floodFill 实现区域生长
    """
    h, w = img.shape
    # floodFill 需要的 mask 尺寸必须是 (h+2, w+2)
    mask = np.zeros((h + 2, w + 2), np.uint8)
    res_img = img.copy()

    # loDiff 和 upDiff 定义了与种子点灰度差的容忍度
    flags = 8 | (255 << 8) | cv2.FLOODFILL_MASK_ONLY
    cv2.floodFill(res_img, mask, seed_point, 255, 
                  loDiff=tolerance, upDiff=tolerance, flags=flags)

    return mask[1:-1, 1:-1]

# ==========================================
# 2. 标准 K-Means 与 改进的空间加权 K-Means
# ==========================================
def kmeans_standard(img, K):
    """标准的基于灰度的 K-Means"""
    pixel_values = img.reshape((-1, 1)).astype(np.float32)
    criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 100, 0.2)

    _, labels, centers = cv2.kmeans(pixel_values, K, None, criteria, 10, cv2.KMEANS_RANDOM_CENTERS)

    return labels.reshape(img.shape), centers

def kmeans_spatial_weighted(img, K, spatial_weight=1.0):
    """
    改进的 K-Means：引入坐标 (x,y) 权重
    将特征从 [灰度] 扩展为 [灰度, weight*X, weight*Y]
    """
    h, w = img.shape
    x, y = np.meshgrid(np.arange(w), np.arange(h))

    # 极度重要：必须将灰度和坐标都归一化到 0~1，否则坐标数值过大会吞噬灰度特征
    img_norm = img.astype(np.float32) / 255.0
    x_norm = x.astype(np.float32) / w
    y_norm = y.astype(np.float32) / h

    # 拼接特征向量
    features = np.vstack((
        img_norm.flatten(), 
        (x_norm * spatial_weight).flatten(), 
        (y_norm * spatial_weight).flatten()
    )).T

    criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 100, 0.2)
    _, labels, _ = cv2.kmeans(features, K, None, criteria, 10, cv2.KMEANS_RANDOM_CENTERS)

    return labels.reshape(img.shape)

# ==========================================
# 3. 拓展：基于距离变换的分水岭算法
# ==========================================
def distance_transform_watershed(img):
    """
    使用距离变换生成精确 Mark 的分水岭算法
    """
    # 1. 初始二值化 (Otsu)
    _, thresh = cv2.threshold(img, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

    # 2. 形态学开运算去除噪点
    kernel = np.ones((3,3), np.uint8)
    opening = cv2.morphologyEx(thresh, cv2.MORPH_OPEN, kernel, iterations=2)

    # 3. 确定背景区域 (膨胀)
    sure_bg = cv2.dilate(opening, kernel, iterations=3)

    # 4. 距离变换 (核心拓展步) -> 确定前景区域
    dist_transform = cv2.distanceTransform(opening, cv2.DIST_L2, 5)
    # 设定距离阈值，得到确定的病灶核心前景
    _, sure_fg = cv2.threshold(dist_transform, 0.6 * dist_transform.max(), 255, 0)
    sure_fg = np.uint8(sure_fg)

    # 5. 确定未知区域 (边界)
    unknown = cv2.subtract(sure_bg, sure_fg)

    # 6. 标记连通域
    _, markers = cv2.connectedComponents(sure_fg)
    markers = markers + 1         # 所有标记加 1，保证背景是 1 而不是 0
    markers[unknown == 255] = 0   # 未知边界区域标记为 0

    # 7. 实施分水岭
    # watershed 需要 3 通道图像
    img_color = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
    markers = cv2.watershed(img_color, markers)

    # 提取病灶掩膜 (通常最大的高亮区连通域，这里简化提取确定的前景)
    lesion_mask = np.zeros_like(img)
    lesion_mask[markers > 1] = 255

    return dist_transform, markers, lesion_mask

def extract_lesion_from_labels(labels, img):
    """工具函数：从聚类 Label 中自动提取高亮病灶区"""
    # T2 中病灶通常灰度最高，找出平均灰度最高的聚类簇
    num_clusters = np.max(labels) + 1
    max_mean = -1
    best_label = 0
    for i in range(num_clusters):
        mean_val = np.mean(img[labels == i])
        if mean_val > max_mean:
            max_mean = mean_val
            best_label = i
    mask = np.zeros_like(img)
    mask[labels == best_label] = 255
    return mask

def get_kmeans_mask(img, k, w_spatial, seed_x, seed_y):
    """手动实现改进 K-means：灰度 + 空间坐标"""
    h, w = img.shape
    y_idx, x_idx = np.indices((h, w))
    # 构建特征向量：[灰度, 归一化x*权重, 归一化y*权重]
    data = np.stack([
        img.flatten() / 255.0, 
        (x_idx.flatten() / w) * w_spatial, 
        (y_idx.flatten() / h) * w_spatial
    ], axis=1).astype(np.float32)

    criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 10, 1.0)
    _, labels, _ = cv2.kmeans(data, k, None, criteria, 10, cv2.KMEANS_PP_CENTERS)
    labels = labels.reshape((h, w))

    # 根据种子点坐标提取对应的聚类簇
    target_label = labels[seed_y, seed_x]
    mask = np.zeros_like(img)
    mask[labels == target_label] = 255
    return mask, labels
